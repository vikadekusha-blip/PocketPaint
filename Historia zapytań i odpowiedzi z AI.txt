import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.ArrayList;
import java.util.List;

// ==========================================
// 1. ARCHITEKTURA FIGUR (OOP: Hierarchia i Polimorfizm)
// ==========================================

// Klasa abstrakcyjna dla wszystkich figur
abstract class Shape {
    protected Color color;
    protected int strokeWidth;

    public Shape(Color color, int strokeWidth) {
        this.color = color;
        this.strokeWidth = strokeWidth;
    }

    // Każda figura sama wie, jak się narysować
    public abstract void draw(Graphics2D g2d);
}

// Linia (do swobodnego rysowania i linii prostych)
class LineShape extends Shape {
    private int x1, y1, x2, y2;

    public LineShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x1 = x1;
        this.y1 = y1;
        this.x2 = x2;
        this.y2 = y2;
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2d.drawLine(x1, y1, x2, y2);
    }
}

// Prostokąt
class RectangleShape extends Shape {
    private int x, y, width, height;

    public RectangleShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x = Math.min(x1, x2);
        this.y = Math.min(y1, y2);
        this.width = Math.abs(x1 - x2);
        this.height = Math.abs(y1 - y2);
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth));
        g2d.drawRect(x, y, width, height);
    }
}

// Owal / Koło
class OvalShape extends Shape {
    private int x, y, width, height;

    public OvalShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x = Math.min(x1, x2);
        this.y = Math.min(y1, y2);
        this.width = Math.abs(x1 - x2);
        this.height = Math.abs(y1 - y2);
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth));
        g2d.drawOval(x, y, width, height);
    }
}

// ==========================================
// 2. PŁÓTNO DO RYSOWANIA (Paint Panel)
// ==========================================
class PaintPanel extends JPanel {
    private List<Shape> shapes = new ArrayList<>(); // Historia figur (Undo / Lista)
    
    // Aktualne ustawienia pędzla
    private String currentTool = "Brush"; // Brush, Line, Rect, Oval, Eraser
    private Color currentColor = Color.BLACK;
    private int currentStroke = 5;

    // Współrzędne myszy do śledzenia
    private int lastX, lastY;
    private int startX, startY;
    private Shape previewShape = null; // Do wyświetlania figury w trakcie rozciągania

    public PaintPanel() {
        setBackground(Color.WHITE);

        // Obsługa kliknięć i naciśnięć myszy
        addMouseListener(new MouseAdapter() {
            @Override
            public void mousePressed(MouseEvent e) {
                startX = e.getX();
                startY = e.getY();
                lastX = startX;
                lastY = startY;
            }

            @Override
            public void mouseReleased(MouseEvent e) {
                // Kiedy puszczamy mysz — ostatecznie dodajemy figurę do historii
                if (previewShape != null) {
                    shapes.add(previewShape);
                    previewShape = null;
                    repaint();
                }
            }
        });

        // Obsługa ruchu myszy z wciśniętym przyciskiem
        addMouseMotionListener(new MouseMotionAdapter() {
            @Override
            public void mouseDragged(MouseEvent e) {
                int currentX = e.getX();
                int currentY = e.getY();

                Color activeColor = currentTool.equals("Eraser") ? Color.WHITE : currentColor;

                switch (currentTool) {
                    case "Brush":
                    case "Eraser":
                        // Swobodne rysowanie — to mnóstwo małych połączonych linii
                        shapes.add(new LineShape(lastX, lastY, currentX, currentY, activeColor, currentStroke));
                        lastX = currentX;
                        lastY = currentY;
                        break;
                    case "Line":
                        previewShape = new LineShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                    case "Rect":
                        previewShape = new RectangleShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                    case "Oval":
                        previewShape = new OvalShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                }
                repaint(); // Przerysowujemy płótno
            }
        });
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        
        // Włączamy wygładzanie, aby linie nie były pikselowymi kostkami
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // Rysujemy wszystkie zapisane figury (Polimorfizm w działaniu)
        for (Shape shape : shapes) {
            shape.draw(g2d);
        }

        // Rysujemy figurę, którą w tym momencie rozciągamy myszką
        if (previewShape != null) {
            previewShape.draw(g2d);
        }
    }

    // Metody sterowania dla panelu narzędzi
    public void setTool(String tool) { this.currentTool = tool; }
    public void setColor(Color color) { this.currentColor = color; }
    public void setStroke(int stroke) { this.currentStroke = stroke; }

    public void clearCanvas() {
        shapes.clear();
        previewShape = null;
        repaint();
    }

    public void undo() {
        if (!shapes.isEmpty()) {
            shapes.remove(shapes.size() - 1);
            repaint();
        }
    }
}

// ==========================================
// 3. GŁÓWNE OKNO APLIKACJI (GUI)
// ==========================================
public class PocketPaint extends JFrame {
    private PaintPanel paintPanel;

    public PocketPaint() {
        setTitle("Pocket Paint");
        setSize(900, 600);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        paintPanel = new PaintPanel();
        add(paintPanel, BorderLayout.CENTER);

        // Tworzymy panel sterowania na górze
        JPanel controlPanel = new JPanel();
        controlPanel.setLayout(new FlowLayout(FlowLayout.LEFT, 10, 10));
        controlPanel.setBackground(new Color(240, 240, 240));

        // Wybór narzędzi
        String[] tools = {"Brush", "Line", "Rect", "Oval", "Eraser"};
        JComboBox<String> toolBox = new JComboBox<>(tools);
        toolBox.addActionListener(e -> paintPanel.setTool((String) toolBox.getSelectedItem()));
        controlPanel.add(new JLabel("Narzędzie:"));
        controlPanel.add(toolBox);

        // Grubość linii (Suwak)
        JSlider strokeSlider = new JSlider(1, 30, 5);
        strokeSlider.addChangeListener(e -> paintPanel.setStroke(strokeSlider.getValue()));
        controlPanel.add(new JLabel("Rozmiar:"));
        controlPanel.add(strokeSlider);

        // Wybór koloru
        JButton colorBtn = new JButton(" Wybierz Kolor ");
        colorBtn.setBackground(Color.BLACK);
        colorBtn.setForeground(Color.WHITE);
        colorBtn.addActionListener(e -> {
            Color selected = JColorChooser.showDialog(this, "Wybierz Kolor Pędzla", Color.BLACK);
            if (selected != null) {
                paintPanel.setColor(selected);
                colorBtn.setBackground(selected);
                // Zmieniamy kolor tekstu na przycisku dla czytelności
                double luminance = (0.299 * selected.getRed() + 0.587 * selected.getGreen() + 0.114 * selected.getBlue()) / 255;
                colorBtn.setForeground(luminance > 0.5 ? Color.BLACK : Color.WHITE);
            }
        });
        controlPanel.add(colorBtn);

        // Przycisk Undo (Cofnij)
        JButton undoBtn = new JButton("Cofnij (Ctrl+Z)");
        undoBtn.addActionListener(e -> paintPanel.undo());
        controlPanel.add(undoBtn);

        // Przycisk Clear (Wyczyść)
        JButton clearBtn = new JButton("Wyczyść");
        clearBtn.addActionListener(e -> paintPanel.clearCanvas());
        controlPanel.add(clearBtn);

        add(controlPanel, BorderLayout.NORTH);

        // Dodajemy skrót klawiszowy Ctrl+Z dla Undo
        paintPanel.getInputMap(JComponent.WHEN_IN_FOCUSED_WINDOW).put(
                KeyStroke.getKeyStroke(KeyEvent.VK_Z, InputEvent.CTRL_DOWN_MASK), "undoAction"
        );
        paintPanel.getActionMap().put("undoAction", new AbstractAction() {
            @Override
            public void actionPerformed(ActionEvent e) {
                paintPanel.undo();
            }
        });
    }

    public static void main(String[] args) {
        // Uruchomienie interfejsu w bezpiecznym wątku Swing
        SwingUtilities.invokeLater(() -> {
            new PocketPaint().setVisible(true);
        });
    }
}

Czystość kodu przed wykładowcą (OOP na maksa): W wymaganiach projektu masz wyraźnie napisane o użyciu zaawansowanych mechanizmów OOP (dziedziczenie, polimorfizm, enkapsulacja). Kiedy sprawdzający otworzy projekt i zobaczy estetyczną strukturę, gdzie każda klasa odpowiada za swoje zadanie (jeden plik — jedna figura, inny — płótno), od razu jest to dobry wskaźnik. Wygodne czytanie i modyfikacja: Wyobraź sobie, że chcesz dodać nową figurę (np. trójkąt). Jeśli masz wszystko w jednym pliku, będziesz musiała przewijać setki linii kodu i wcinać się w czyjąś logikę. W opcji z podziałem po prostu tworzysz plik TriangleShape.java, piszesz co trzeba i o nic się nie martwisz.

Łatwiejsze szukanie błędów: Jeśli popsuje się logika rysowania na płótnie — od razu idziesz do PaintPanel.java. Jeśli błąd jest w interfejsie lub przyciskach — zaglądasz do PocketPaint.java. Wszystko jest odizolowane.

// Klasa abstrakcyjna dla wszystkich figur (OOP: Enkapsulacja)
public abstract class Shape {
    protected Color color;
    protected int strokeWidth;

    public Shape(Color color, int strokeWidth) {
        this.color = color;
        this.strokeWidth = strokeWidth;
    }

    // Każda figura będzie nadpisywać tę metodę pod siebie (OOP: Polimorfizm)
    public abstract void draw(Graphics2D g2d);
}

Java
import java.awt.*;

// Linia do swobodnego rysowania i linii prostych (OOP: Dziedziczenie)
class LineShape extends Shape {
    private int x1, y1, x2, y2;

    public LineShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x1 = x1;
        this.y1 = y1;
        this.x2 = x2;
        this.y2 = y2;
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        // Zaokrąglamy końce linii, aby pędzel rysował płynnie, bez poszarpanych krawędzi
        g2d.setStroke(new BasicStroke(strokeWidth, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2d.drawLine(x1, y1, x2, y2);
    }
}

// Prostokąt (OOP: Dziedziczenie)
class RectangleShape extends Shape {
    private int x, y, width, height;

    public RectangleShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        // Math.min i Math.abs są potrzebne, aby figura rysowała się poprawnie w każdą stronę przeciągania myszy
        this.x = Math.min(x1, x2);
        this.y = Math.min(y1, y2);
        this.width = Math.abs(x1 - x2);
        this.height = Math.abs(y1 - y2);
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth));
        g2d.drawRect(x, y, width, height);
    }
}

// Owal / Koło (OOP: Dziedziczenie) 
class OvalShape extends Shape {
    private int x, y, width, height;

    public OvalShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x = Math.min(x1, x2);
        this.y = Math.min(y1, y2);
        this.width = Math.abs(x1 - x2);
        this.height = Math.abs(y1 - y2);
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth));
        g2d.drawOval(x, y, width, height);
    }
}

To nasze cyfrowe płótno. Łapie ruchy myszki, przechowuje listę wszystkich narysowanych figur i zajmuje się ich renderowaniem.

Java

Java
import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.ArrayList;
import java.util.List;

public class PaintPanel extends JPanel {
    // Lista do przechowywania historii wszystkich narysowanych obiektów (figur)
    private List<Shape> shapes = new ArrayList<>(); 
    
    // Domyślne ustawienia narzędzia
    private String currentTool = "Brush"; 
    private Color currentColor = Color.BLACK;
    private int currentStroke = 5;

    // Zmienne pomocnicze dla myszy
    private int lastX, lastY;
    private int startX, startY;
    private Shape previewShape = null; // Figura, którą "ciągniemy" do momentu puszczenia przycisku myszy

    public PaintPanel() {
        setBackground(Color.WHITE); // Początkowo płótno jest czyste i białe

        // Słuchacz kliknięć myszy (MouseListener)
        addMouseListener(new MouseAdapter() {
            @Override
            public void mousePressed(MouseEvent e) {
                startX = e.getX();
                startY = e.getY();
                lastX = startX;
                lastY = startY;
            }

            @Override
            public void mouseReleased(MouseEvent e) {
                // Przycisk zwolniony — zatwierdzamy figurę na płótnie
                if (previewShape != null) {
                    shapes.add(previewShape);
                    previewShape = null;
                    repaint();
                }
            }
        });

        // Słuchacz ruchu myszy (MouseMotionListener)
        addMouseMotionListener(new MouseMotionAdapter() {
            @Override
            public void mouseDragged(MouseEvent e) {
                int currentX = e.getX();
                int currentY = e.getY();

                // Jeśli wybrana jest gumka, po prostu rysujemy kolorem tła (białym)
                Color activeColor = currentTool.equals("Eraser") ? Color.WHITE : currentColor;

                switch (currentTool) {
                    case "Brush":
                    case "Eraser":
                        // Swobodne rysowanie to nieprzerwana seria małych linii
                        shapes.add(new LineShape(lastX, lastY, currentX, currentY, activeColor, currentStroke));
                        lastX = currentX;
                        lastY = currentY;
                        break;
                    case "Line":
                        previewShape = new LineShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                    case "Rect":
                        previewShape = new RectangleShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                    case "Oval":
                        previewShape = new OvalShape(startX, startY, currentX, currentY, activeColor, currentStroke);
                        break;
                }
                repaint(); // Zmuszamy panel do odświeżenia i wywołania paintComponent
            }
        });
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        
        // Włączamy antyaliasing (wygładzanie), aby rysunki były gładkie i ładne
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // Magia POLIMORFIZMU: wywołujemy .draw() na każdej figurze, a one same wiedzą, jak się rysować!
        for (Shape shape : shapes) {
            shape.draw(g2d);
        }

        // Rysujemy podgląd konturu figury, którą aktualnie rozciągamy
        if (previewShape != null) {
            previewShape.draw(g2d);
        }
    }

    // Metody sterowania dla przycisków interfejsu
    public void setTool(String tool) { this.currentTool = tool; }
    public void setColor(Color color) { this.currentColor = color; }
    public void setStroke(int stroke) { this.currentStroke = stroke; }

    // Czyszczenie płótna
    public void clearCanvas() {
        shapes.clear();
        previewShape = null;
        repaint();
    }

    // Odmiana operacji (Undo)
    public void undo() {
        if (!shapes.isEmpty()) {
            shapes.remove(shapes.size() - 1); // Usuwamy ostatnią figurę z listy
            repaint();
        }
    }
}

1. Plik Shape.java (Abstrakcyjny fundament)
To najważniejsza klasa pod kątem OOP. Nie możemy stworzyć „po prostu figury”, bo nie wiadomo, jak ona wygląda. Dlatego klasa została oznaczona jako abstract.

Java
public abstract class Shape {
    protected Color color;
    protected int strokeWidth;

    public Shape(Color color, int strokeWidth) {
        this.color = color;
        this.strokeWidth = strokeWidth;
    }

    public abstract void draw(Graphics2D g2d);
}
protected Color color i strokeWidth — to pola, w których przechowywane są kolor oraz grubość linii. Modyfikator protected (enkapsulacja) oznacza, że zmienne te są ukryte przed światem zewnętrznym, ale klasy pochodne (linia, koło) mają do nich bezpośredni dostęp.

Konstruktor Shape(...) — kiedy będziemy tworzyć jakąkolwiek figurę, mamy obowiązek przekazać jej kolor i grubość.

public abstract void draw(Graphics2D g2d) — to metoda abstrakcyjna. Nie ma ona ciała (kodu wewnątrz). To po prostu rozkaz dla wszystkich przyszłych figur: „Każdy, kto po mnie dziedziczy, musi napisać swoją metodę draw i wyjaśnić Javie, jak dokładnie ma siebie narysować”.

class LineShape extends Shape {
    private int x1, y1, x2, y2;

    public LineShape(int x1, int y1, int x2, int y2, Color color, int strokeWidth) {
        super(color, strokeWidth);
        this.x1 = x1; this.y1 = y1; this.x2 = x2; this.y2 = y2;
    }

    @Override
    public void draw(Graphics2D g2d) {
        g2d.setColor(color);
        g2d.setStroke(new BasicStroke(strokeWidth, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));

super(color, strokeWidth) — wywołuje konstruktor klasy nadrzędnej Shape, aby przekazać mu wspólne ustawienia.

@Override — adnotacja, która mówi: „Nadpisuję metodę draw z klasy rodzica”.

g2d.setStroke(...) — konfiguruje pędzel. Parametry CAP_ROUND i JOIN_ROUND sprawiają, że końce linii są zaokrąglone. Dzięki temu swobodne rysowanie odręczne wygląda gładko, a nie jak połamane, kwadratowe piksele.

Prostokąt i Owal (RectangleShape i OvalShape)
W Java Swing funkcje drawRect i drawOval przyjmują na wejściu: x oraz y (lewy górny róg), a także width (szerokość) i height (wysokość). Jednak użytkownik może przeciągać myszkę w dowolną stronę — z lewej na prawą, z prawej na lewą, z dołu do góry.

Żeby kształt się nie psuł i nie znikał, w konstruktorze zastosowano odrobinę matematyki:
